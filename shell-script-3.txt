ping test using for loop:
-------------------------

#!/bin/bash

for server in google.com facebook.com codekerdos.in
do
        if ping -W2 -c1 $server &> /dev/null
        then
            echo "$server is reachable"
        else
             echo "$server is not reachable"
        fi
done



muliplication-table.sh:
-----------------------
#!/bin/bash
for (( i=1; i<=10; i++ ))
do
        echo "value of i is $i"
        for (( j=1; j<=20; j++ ))
        do
                echo "$i * $j = $(( $i * $j )) "
        done
done
